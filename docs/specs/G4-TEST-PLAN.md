[project]
name = "g4-test"

EOF

echo 'def foo(): pass' > g4-test/src/a.py
echo 'from src.a import foo' > g4-test/src/b.py

python3 -m neuralmind build g4-test
# Verify: graph has edge a.py → b.py

python3 -m neuralmind build g4-test
# Verify: no dangling edge to a.py in graph.json
